Script SQL de Criação das Tabelas

Basta colar o código abaixo no SQL Editor do seu Supabase Dashboard para criar as tabelas, índices de unicidade e políticas de segurança RLS: 

CREATE TABLE IF NOT EXISTS categories (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name TEXT NOT NULL UNIQUE,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS gifts (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name TEXT NOT NULL,
  description TEXT,
  category_id UUID REFERENCES categories(id) ON DELETE SET NULL,
  image_url TEXT,
  is_active BOOLEAN NOT NULL DEFAULT true,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS reservations (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  gift_id UUID NOT NULL REFERENCES gifts(id) ON DELETE CASCADE,
  guest_name TEXT NOT NULL,
  guest_whatsapp TEXT NOT NULL,
  message TEXT,
  status TEXT NOT NULL DEFAULT 'active' CHECK (status IN ('active', 'cancelled')),
  created_at TIMESTAMPTZ DEFAULT NOW(),
  cancelled_at TIMESTAMPTZ
);

ALTER TABLE public.categories ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.gifts ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.reservations ENABLE ROW LEVEL SECURITY;

DO $$
DECLARE policy_record RECORD;
BEGIN
  FOR policy_record IN
    SELECT tablename, policyname FROM pg_policies
    WHERE schemaname = 'public' AND tablename IN ('categories', 'gifts', 'reservations')
  LOOP
    EXECUTE format('DROP POLICY %I ON public.%I', policy_record.policyname, policy_record.tablename);
  END LOOP;
END $$;

REVOKE ALL ON public.categories, public.gifts, public.reservations FROM anon, authenticated;
GRANT SELECT ON public.categories, public.gifts TO anon, authenticated;
CREATE POLICY public_read_categories ON public.categories FOR SELECT TO anon, authenticated USING (true);
CREATE POLICY public_read_gifts ON public.gifts FOR SELECT TO anon, authenticated USING (true);

CREATE OR REPLACE VIEW public.public_active_gift_reservations AS
  SELECT DISTINCT gift_id FROM public.reservations WHERE status = 'active';
GRANT SELECT ON public.public_active_gift_reservations TO anon, authenticated;

CREATE OR REPLACE FUNCTION public.create_gift_reservation(
  p_gift_id uuid,
  p_guest_name text,
  p_guest_whatsapp text,
  p_message text DEFAULT ''
)
RETURNS jsonb
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public, pg_temp
AS $$
DECLARE
  active_gift boolean;
  created_reservation public.reservations;
BEGIN
  SELECT is_active INTO active_gift FROM public.gifts WHERE id = p_gift_id FOR UPDATE;
  IF NOT FOUND OR NOT active_gift THEN
    RAISE EXCEPTION 'Presente indisponível';
  END IF;
  IF length(trim(p_guest_name)) NOT BETWEEN 1 AND 120
     OR length(trim(p_guest_whatsapp)) NOT BETWEEN 5 AND 30
     OR length(coalesce(p_message, '')) > 1000 THEN
    RAISE EXCEPTION 'Dados da reserva inválidos';
  END IF;
  IF EXISTS (SELECT 1 FROM public.reservations WHERE gift_id = p_gift_id AND status = 'active') THEN
    RAISE EXCEPTION 'Presente já reservado' USING ERRCODE = '23505';
  END IF;
  INSERT INTO public.reservations (gift_id, guest_name, guest_whatsapp, message)
  VALUES (p_gift_id, trim(p_guest_name), trim(p_guest_whatsapp), coalesce(p_message, ''))
  RETURNING * INTO created_reservation;
  RETURN to_jsonb(created_reservation);
END;
$$;
REVOKE ALL ON FUNCTION public.create_gift_reservation(uuid, text, text, text) FROM PUBLIC;
GRANT EXECUTE ON FUNCTION public.create_gift_reservation(uuid, text, text, text) TO anon, authenticated;
